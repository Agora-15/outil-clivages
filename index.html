using UnityEngine; public class ControleCube : MonoBehaviour { [Header("Paramètres de Rotation")] [Tooltip("Vitesse de rotation du cube")] public float vitesseRotation = 100f; void Update() { // 1. Récupération des entrées utilisateur (Flèches Gauche/Droite, touches Q/D, ou Joystick) float inputHorizontal = Input.GetAxis("Horizontal"); // Axe horizontal (Y) float inputVertical = Input.GetAxis("Vertical"); // Axe vertical (X) // 2. Détection de l'orientation du cube dans l'espace // On calcule le produit scalaire entre le "Haut" du cube et le "Haut" du monde // Si le résultat est négatif, cela signifie que le cube est incliné à plus de 90° (tête en bas) bool estALenvers = Vector3.Dot(transform.up, Vector3.up) < 0; // 3. Correction des inversions si le cube est à l'envers if (estALenvers) { // Inverse les commandes pour que l'action reste naturelle à l'écran inputHorizontal *= -1f; inputVertical *= -1f; } // 4. Application des rotations sur le cube // Rotation autour de l'axe X (haut/bas) et de l'axe Y (gauche/droite) float rotationX = inputVertical * vitesseRotation * Time.deltaTime; float rotationY = -inputHorizontal * vitesseRotation * Time.deltaTime; // On applique la rotation dans le repère global (Space.World) // pour éviter que les axes locaux ne se mélangent (phénomène de Gimbal Lock) transform.Rotate(Vector3.up, rotationY, Space.World); transform.Rotate(Vector3.right, rotationX, Space.World); } }
